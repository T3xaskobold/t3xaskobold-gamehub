export default async function handler(req, res) { res.setHeader("Cache-Control", "no-store"); if (req.method !== "GET") { return res.status(405).json({ error: "Methode nicht erlaubt" }); } const clientId = process.env.TWITCH_CLIPS_CLIENT_ID; const clientSecret = process.env.TWITCH_CLIPS_CLIENT_SECRET; if (!clientId || !clientSecret) { return res.status(500).json({ error: "Twitch-Zugangsdaten fehlen" }); } try { // Twitch-Zugriffstoken anfordern const tokenResponse = await fetch( "https://id.twitch.tv/oauth2/token", { method: "POST", headers: { "Content-Type": "application/x-www-form-urlencoded" }, body: new URLSearchParams({ client_id: clientId, client_secret: clientSecret, grant_type: "client_credentials" }) } ); if (!tokenResponse.ok) { throw new Error("Twitch-Anmeldung fehlgeschlagen"); } const tokenData = await tokenResponse.json(); // Prüfen, ob T3xaskobold live ist const streamUrl = new URL( "https://api.twitch.tv/helix/streams" ); streamUrl.searchParams.set( "user_login", "t3xaskobold" ); const streamResponse = await fetch( streamUrl.toString(), { headers: { "Client-ID": clientId, "Authorization": "Bearer " + tokenData.access_token } } ); if (!streamResponse.ok) { throw new Error( "Twitch-LIVE-Status konnte nicht geladen werden" ); } const streamData = await streamResponse.json(); const stream = Array.isArray(streamData.data) ? streamData.data[0] : null; const isLive = Boolean( stream && stream.type === "live" ); // Nur öffentliche Stream-Informationen zurückgeben return res.status(200).json({ live: isLive, channel: "t3xaskobold", title: isLive ? stream.title : null, game: isLive ? stream.game_name : null, viewers: isLive ? stream.viewer_count : null, url: "https://www.twitch.tv/t3xaskobold" }); } catch (error) { console.error( "Fehler bei Twitch-LIVE-Abfrage:", error.message ); return res.status(502).json({ error: "LIVE-Status momentan nicht verfügbar" }); } }
